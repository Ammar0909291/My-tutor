def match(name, keys):
    n = name.lower().strip()
    for k in keys.split(';'):
        k = k.strip().lower()
        if not k: continue
        if k.startswith('='):
            if n == k[1:] or n.startswith(k[1:] + ' (') and False: return True
            if n == k[1:]: return True
            # allow truncated exact like '=function (set' : prefix match when key contains '('
            if '(' in k and n.startswith(k[1:]): return True
        else:
            if n.startswith(k): return True
    return False
out = []
for r in rows:
    d = r['domain']; default, E, H = R[d]
    n = r['name']
    if match(n, E): t = 'E'
    elif match(n, H): t = 'H'
    else: t = default
    # geom default E, arith default H when unmatched
    out.append(dict(r, tier=t))
json.dump(out, open('tiers.json', 'w'))
